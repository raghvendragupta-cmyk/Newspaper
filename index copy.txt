  <!-- =========================================
       5. JAVASCRIPT LOGIC & CHARTS ENGINE
       ========================================= -->
  <script>
    let currentLeaf = 0;
    const totalLeaves = 4;
    const chartConfigs = {};
    let modalChartInstance = null;

    function updateZIndex() {
      for (let i = 1; i <= totalLeaves; i++) {
        const leaf = document.getElementById(`leaf-${i}`);
        if (i <= currentLeaf) {
          leaf.style.zIndex = i;
        } else {
          leaf.style.zIndex = totalLeaves - i + 1;
        }
      }
      
      document.getElementById('prev-btn').disabled = (currentLeaf === 0);
      document.getElementById('next-btn').disabled = (currentLeaf === totalLeaves);
      document.getElementById('spread-label').innerText = `Spread ${currentLeaf + 1} of ${totalLeaves}`;
    }
    updateZIndex();

    function flipNext() {
      if (currentLeaf < totalLeaves) {
        currentLeaf++;
        const leaf = document.getElementById(`leaf-${currentLeaf}`);
        leaf.style.zIndex = 100;
        leaf.classList.add('flipped');
        setTimeout(updateZIndex, 800);
      }
    }

    function flipPrev() {
      if (currentLeaf > 0) {
        const leaf = document.getElementById(`leaf-${currentLeaf}`);
        leaf.style.zIndex = 100;
        leaf.classList.remove('flipped');
        currentLeaf--;
        setTimeout(updateZIndex, 800);
      }
    }

    function zoomCard(cardElem) {
      const modal = document.getElementById('mag-modal');
      const target = document.getElementById('modal-body-target');
      
      if (modalChartInstance) {
        modalChartInstance.destroy();
        modalChartInstance = null;
      }

      const canvasElem = cardElem.querySelector('canvas');
      if (canvasElem && chartConfigs[canvasElem.id]) {
        const config = chartConfigs[canvasElem.id];
        const titleHTML = cardElem.querySelector('.section-title')?.outerHTML || '';
        
        target.innerHTML = `${titleHTML}<div style="height: 320px; width: 100%; margin-top: 15px;"><canvas id="modal-canvas"></canvas></div>`;
        modal.style.display = 'flex';
        
        const ctx = document.getElementById('modal-canvas');
        modalChartInstance = new Chart(ctx, {
          type: config.type,
          data: JSON.parse(JSON.stringify(config.data)),
          options: { ...config.options, responsive: true, maintainAspectRatio: false }
        });
      } else {
        target.innerHTML = `<div style="font-size: 110%; line-height: 1.6;">${cardElem.innerHTML}</div>`;
        modal.style.display = 'flex';
      }
    }

    function closeModal() {
      document.getElementById('mag-modal').style.display = 'none';
    }

    let chartsDrawn = false;

    function openBook() {
      document.getElementById('teaser-screen').style.opacity = '0';
      setTimeout(() => {
        document.getElementById('teaser-screen').style.display = 'none';
        document.getElementById('book-viewport').style.display = 'flex';
        if (!chartsDrawn) setTimeout(renderCharts, 150);
      }, 500);
    }

    function closeBook() {
      while(currentLeaf > 0) flipPrev();
      document.getElementById('book-viewport').style.display = 'none';
      document.getElementById('teaser-screen').style.display = 'flex';
      setTimeout(() => { document.getElementById('teaser-screen').style.opacity = '1'; }, 50);
    }

    function createChart(id, type, data, options) {
      chartConfigs[id] = { type, data, options };
      const ctx = document.getElementById(id);
      if (ctx) {
        return new Chart(ctx, { type, data, options: { ...options, responsive: true, maintainAspectRatio: false } });
      }
    }

    function renderCharts() {
